cmake_minimum_required(VERSION 3.25)
project(stackpulse VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STACKPULSE_BUILD_CLIENT "Build native SDL Classic client and input diagnostic" ON)
option(STACKPULSE_BUILD_TESTS "Build Catch2 tests" ON)
option(STACKPULSE_FETCH_DEPENDENCIES "Fetch pinned source dependencies" ON)
include(FetchContent)
function(stackpulse_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
add_library(stackpulse_core src/core/core.cpp src/core/classic.cpp)
target_include_directories(stackpulse_core PUBLIC include)
stackpulse_warnings(stackpulse_core)
add_library(stackpulse_replay src/replay/replay.cpp)
target_link_libraries(stackpulse_replay PUBLIC stackpulse_core)
stackpulse_warnings(stackpulse_replay)
add_executable(stackpulse_replay_verify tools/replay_verify.cpp)
target_link_libraries(stackpulse_replay_verify PRIVATE stackpulse_replay)
stackpulse_warnings(stackpulse_replay_verify)
add_executable(stackpulse_headless tools/headless.cpp)
target_link_libraries(stackpulse_headless PRIVATE stackpulse_core)
stackpulse_warnings(stackpulse_headless)
add_library(stackpulse_input src/platform/input.cpp)
target_include_directories(stackpulse_input PUBLIC include)
stackpulse_warnings(stackpulse_input)
if(STACKPULSE_BUILD_CLIENT)
  if(STACKPULSE_FETCH_DEPENDENCIES)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
      GIT_TAG 7f3ae3d57459e59943a4ecfefc8f6277ec6bf540)
    FetchContent_MakeAvailable(SDL3)
  else()
    find_package(SDL3 3.2 CONFIG REQUIRED)
  endif()
  add_executable(stackpulse_client apps/client/main.cpp)
  target_link_libraries(stackpulse_client PRIVATE stackpulse_input stackpulse_core stackpulse_replay SDL3::SDL3)
  stackpulse_warnings(stackpulse_client)
  add_executable(stackpulse_diagnostic apps/client/diagnostic.cpp)
  target_link_libraries(stackpulse_diagnostic PRIVATE stackpulse_input SDL3::SDL3)
  stackpulse_warnings(stackpulse_diagnostic)
endif()
if(STACKPULSE_BUILD_TESTS)
  enable_testing()
  if(STACKPULSE_FETCH_DEPENDENCIES)
    FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git
      GIT_TAG 56809e5282f104c5c8b570e7c2996cdc352d94f1)
    FetchContent_MakeAvailable(Catch2)
  else()
    find_package(Catch2 3 CONFIG REQUIRED)
  endif()
  add_executable(stackpulse_tests tests/core/core_tests.cpp tests/core/classic_tests.cpp tests/platform/input_tests.cpp tests/replay/replay_tests.cpp)
  target_link_libraries(stackpulse_tests PRIVATE stackpulse_core stackpulse_replay stackpulse_input Catch2::Catch2WithMain)
  stackpulse_warnings(stackpulse_tests)
  add_test(NAME unit COMMAND stackpulse_tests)
  add_test(NAME deterministic_example COMMAND stackpulse_headless)
  set_tests_properties(deterministic_example PROPERTIES PASS_REGULAR_EXPRESSION "hash=feca8ff604d809e5")
  if(STACKPULSE_BUILD_CLIENT)
    add_test(NAME native_smoke COMMAND stackpulse_client --smoke-frames 5)
    set_tests_properties(native_smoke PROPERTIES TIMEOUT 15)
  endif()
endif()
